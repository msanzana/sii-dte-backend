    public function test_factura_33_recorre_automatizacion_completa_hasta_sent(): void
    {
        $this->runFamily(
            dteType: DteType::from(33),
            headerPayload: null,
            expectedSendAction: 'send_factura',
            expectedTransport: 'soap_upload_factura',
            expectedDispatchStatus: 'upload_ok',
            expectedUploadCode: '0',
            expectedTrackId: '0261000001',
        );
    }

    public function test_boleta_39_recorre_automatizacion_completa_hasta_sent(): void
    {
        $this->runFamily(
            dteType: DteType::from(39),
            headerPayload: [
                'ind_servicio' => 3,
            ],
            expectedSendAction: 'send_boleta',
            expectedTransport: 'rest_upload_boleta',
            expectedDispatchStatus: 'sent',
            expectedUploadCode: 'REC',
            expectedTrackId: '32500001',
        );
    }

                    CertificateMaterialExtractorService::class

                    'extractFromRawMaterial'

                )

                    'certificate_pem' =>

                        'Aw==',

            $loadCertificateMaterial =
                new LoadCertificateMaterialForEmisionService(
                    loadCertificateForEmisionService:
                        $loadCertificate,

                    certificateMaterialExtractorService:
                        $certificateExtractor,

                    logRepository:
                        $logRepository,
                );

            /*
             * --------------------------------------------------------------
             * SIGN XML REAL como caso de uso.
             * Solo se simula la primitiva criptografica.
             * --------------------------------------------------------------
             */

            $xmlSignature =
                Mockery::mock(
                    DteXmlSignatureService::class
                );

            $xmlSignature
                ->shouldReceive('signDte')

                        string $xmlWithTed

                        return [
                            'signed_xml' =>

                                'DTE_F77_TEST',

                            'tmst_firma' =>
                                '2026-10-04T10:00:00',
                        ];
                    }
                );

            $signUseCase =
                new SignDteXmlUseCase(
                    documentRepository:
                        $documentRepository,

                    companyRepository:
                        $companyRepository,

                    logRepository:
                        $logRepository,

                    dteXmlSignatureService:
                        $xmlSignature,

                    storageService:
                        $storageService,

                    xmlSignDomainService:
                        new DteXmlSignDomainService(),

                    loadCertificateMaterialForEmisionService:
                        $loadCertificateMaterial,
                );

            /*
             * --------------------------------------------------------------
             * Dispatch repository STATEFUL

                    SiiDispatchRepositoryInterface::class
                );

            $dispatchRepository

                    }
                );

            $dispatchRepository
                ->shouldReceive('update')

                    function (
                        SiiDispatch $updated
                    ) use (&$dispatch): SiiDispatch {

                    }
                );

            /*
             * --------------------------------------------------------------
             * Sobre / firma de sobre
             * --------------------------------------------------------------
             */

            $facturaEnvelope =
                Mockery::mock(

                );

            $facturaEnvelope
                ->shouldReceive('build')
                ->andReturn([
                    'envelope_xml' =>

                    'payload_xml' =>

             */

                        200,

                    'raw_body' =>
                        '<RECEPCIONDTE><STATUS>0</STATUS></RECEPCIONDTE>',

                        '0',

                    'status_message' =>

            /*
             * ==============================================================
             * PASO 1 - BUILD XML
             * ==============================================================
             */

            $step1 =
                $automation->execute(
                    new AdvanceDteAutomationInputDto(

                'build_xml',
                $step1->executeAction
            );

            $this->assertSame(
                DteStatus::XML_BUILT->value,

            );

            $this->assertTrue(
                $step1->shouldRequeueImmediately
            );

            $this->assertNotNull(
                $document->unsignedXmlPath()
            );

            $this->assertBusinessReferences(
                $document
            );

            $baseXml =
                File::get(
                    storage_path(
                        $document->unsignedXmlPath()
                    )
                );

            $this->assertStringContainsString(
                '<TipoDTE>'
                    . $dteType->value
                    . '</TipoDTE>',
                $baseXml
            );

            $this->assertStringContainsString(
                '<Folio>77</Folio>',
                $baseXml
            );

            if (
                $dteType->value === 39
            ) {
                $this->assertStringContainsString(
                    '<IndServicio>3</IndServicio>',
                    $baseXml
                );

                $this->assertStringNotContainsString(
                    '<IndServicio>',
                    $baseXml
                );

             * ==============================================================
             * PASO 2 - BUILD TED
             * ==============================================================
             */

                $automation->execute(
                    new AdvanceDteAutomationInputDto(

                'build_ted',
                $step2->executeAction
            );

                DteStatus::TED_BUILT->value,

                $step2->shouldRequeueImmediately
            );

            $this->assertNotNull(
                $document->tedXml()
            );

            $this->assertStringContainsString(
                '<TD>'
                    . $dteType->value
                    . '</TD>',
                (string) $document->tedXml()
            );

            $this->assertStringContainsString(
                '<F>77</F>',
                (string) $document->tedXml()
            );

            $this->assertStringContainsString(
                '<MNT>47600</MNT>',
                (string) $document->tedXml()
            );

            $this->assertBusinessReferences(
                $document
            );

            /*
             * ==============================================================
             * PASO 3 - SIGN
             * ==============================================================

                $automation->execute(
                    new AdvanceDteAutomationInputDto(

                    )

                'sign_xml',

    private function seedTokenProviderCache(
        object $provider,
        string $environment,
        int $companyId,
        int $certificateId,
        string $token,
    ): void {
        $reflection = new ReflectionClass(
            $provider
        );

        $cacheKeyMethod = null;

        foreach (
            $reflection->getMethods()
            as $method
        ) {
            $methodName =
                strtolower(
                    $method->getName()
                );

            if (
                str_contains(
                    $methodName,
                    'cache'
                )
                && str_contains(
                    $methodName,
                    'key'
                )
                && $method->getNumberOfParameters() === 3
            ) {
                $cacheKeyMethod = $method;
                break;
            }
        }

        if (
            !$cacheKeyMethod instanceof ReflectionMethod
        ) {
            throw new RuntimeException(
                'No fue posible localizar el metodo interno de cache key en '
                . $provider::class
                . '.'
            );
        }

        $cacheKeyMethod->setAccessible(
            true
        );

        $cacheKey =
            $cacheKeyMethod->invoke(
                $provider,
                $environment,
                $companyId,
                $certificateId,
            );

        if (
            !is_string($cacheKey)
            || trim($cacheKey) === ''
        ) {
            throw new RuntimeException(
                'El provider '
                . $provider::class
                . ' no genero una cache key valida.'
            );
        }

        Cache::put(
            $cacheKey,
            $token,
            600
        );
    }

<?php
namespace App\Modules\Dte\Infrastructure\Sii;

use App\Modules\Dte\Infrastructure\Sii\SiiSoapAuthenticationService;
use Illuminate\Contracts\Cache\LockProvider;
use Illuminate\Contracts\Cache\LockTimeoutException;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Crypt;
use RuntimeException;
use Throwable;


final class SiiTokenProviderService
{
    public function __construct(
        private readonly SiiSoapAuthenticationService $authenticationService,
    ){}
    public function get(
        string $environment,
        int $companyId,
        int $certificateId,
        string $privateKeyPem,
        string $certificateBase64,
        string $modulusBase64,
        string $exponentBase64,
    ):array
    {

        $storeName = (string) config(
            'dte.sii.transport.cache_store',
            'file'
        );

        /*
        |--------------------------------------------------------------------------
        | Repository de cache
        |--------------------------------------------------------------------------
        |
        | Lo utilizamos para get(), put(), forget(), etc.
        |
        */
        $cache = Cache::store($storeName);

        /*
        |--------------------------------------------------------------------------
        | Store real
        |--------------------------------------------------------------------------
        |
        | El store subyacente es el que debe implementar LockProvider
        | para poder crear locks atómicos.
        |
        */
        $lockStore = $cache->getStore();

        if (! $lockStore instanceof LockProvider) {
            throw new RuntimeException(
                sprintf(
                    'El cache store "%s" no soporta locks atómicos.',
                    $storeName
                )
            );
        }

        $cacheKey = $this->cacheKey(
            environment: $environment,
            companyId: $companyId,
            certificateId: $certificateId,
        );

        /*
        |--------------------------------------------------------------------------
        | Primero intentamos reutilizar un TOKEN existente
        |--------------------------------------------------------------------------
        */

        $cachedToken = $this->readCachedToken(
            $cache->get($cacheKey)
        );

        if ($cachedToken !== null) {

            return [
                'token' => $cachedToken,
                'source' => 'cache',
            ];
        }

        /*
        |--------------------------------------------------------------------------
        | Si no existe TOKEN, bloqueamos para evitar que varios workers
        | soliciten uno nuevo simultáneamente.
        |--------------------------------------------------------------------------
        */

        try {
            return $lockStore->lock(
                $cacheKey . ':lock',
                90
            )
            ->block(
                30,
                function () use (
                    $cache,
                    $cacheKey,
                    $environment,
                    $privateKeyPem,
                    $certificateBase64,
                    $modulusBase64,
                    $exponentBase64,
                ): array{
                    $cachedToken = $this->readCachedToken(
                        $cache->get($cacheKey)
                    );

                    if ($cachedToken !== null) {
                        return [
                            'token' => $cachedToken,
                            'source' => 'cache',
                        ];
                    }
                    $token = $this->authenticationService
                    ->authenticate(
                        environment: $environment,
                        privateKeyPem: $privateKeyPem,
                        certificateBase64: $certificateBase64,
                        modulusBase64: $modulusBase64,
                        exponentBase64: $exponentBase64,
                    );
                    $ttlMinutes = max(
                        1,
                        (int) config(
                            'dte.sii.transport.token_cache_ttl_minutes',
                            50
                        )
                    );

                    $cache->put(
                        $cacheKey,
                        Crypt::encryptString($token),
                        now()->addMinutes($ttlMinutes)
                    );

                    return [
                        'token' => $token,
                        'source' => 'fresh',
                    ];
                }
           );
       } catch (LockTimeoutException) {
            $cachedToken = $this->readCachedToken(
                $cache->get($cacheKey)
            );

            if ($cachedToken !== null) {
                return [
                    'token' => $cachedToken,
                    'source' => 'cache',
                ];
            }

            throw new RuntimeException(
                'No fue posible obtener el lock para generar el TOKEN SII.'
            );
        }
    }

    public function forget
    (
        string $environment,
        int $companyId,
        int $certificateId,
    ):void {
        Cache::store(
            (string) config(
                'dte.sii.transport.cache_store',
                'file')
            )->forget(
                $this->cacheKey(
                    environment: $environment,
                    companyId: $companyId,
                    certificateId: $certificateId,
            )
        );
    }

    private function cacheKey(
        string $environment,
        int $companyId,
        int $certificateId,
    ): string {
        return sprintf(
            'sii:token:%s:company:%d:certificate:%d',
            strtolower(trim($environment)),
            $companyId,
            $certificateId,
        );
    }

    private function readCachedToken(
        mixed $encryptedToken
    ): ?string {
        if(!is_string($encryptedToken) || trim($encryptedToken) === '')
        {
            return null;
        }
        try {
            $token = Crypt::decryptString(
                $encryptedToken
            );
        } catch (Throwable) {
            return null;
        }

        $token = trim($token);

        return $token === ''
            ? null
            : $token;
    }
}

<?php

namespace App\Modules\Dte\Infrastructure\Sii;
use Illuminate\Support\Facades\Cache;
use Illuminate\Contracts\Cache\LockProvider;
use RuntimeException;
use Illuminate\Contracts\Cache\LockTimeoutException;
final class SiiBoletaTokenProviderService
{
    public function __construct(
        private readonly SiiBoletaApiAuthenticationService $authenticationService,
    ) {
    }

    /**
     * @return array{
     *     token: string,
     *     source: string
     * }
     */
    public function get(
        string $environment,
        int $companyId,
        int $certificateId,
        string $privateKeyPem,
        string $certificateBase64,
        string $modulusBase64,
        string $exponentBase64,
    ): array {
        $cacheKey = $this->cacheKey(
            environment: $environment,
            companyId: $companyId,
            certificateId: $certificateId,
        );

        $cachedToken = Cache::store('file')->get(
            $cacheKey
        );

        if (
            is_string($cachedToken)
            && trim($cachedToken) !== ''
        ) {
            return [
                'token' => trim($cachedToken),
                'source' => 'cache',
            ];
        }
        $lockKey = $cacheKey . ':lock';

        $lockStore = Cache::store('file')->getStore();

        if (!$lockStore instanceof LockProvider) {
            throw new RuntimeException(
                'El store de cache configurado para tokens SII de boleta no soporta locks.'
            );
        }
        try {
            return $lockStore
                ->lock(
                    $lockKey,
                    90
                )
                ->block(
                    30,
                    function () use (
                        $environment,
                        $privateKeyPem,
                        $certificateBase64,
                        $modulusBase64,
                        $exponentBase64,
                        $cacheKey
                    ): array {
                        $cachedTokenAfterLock = Cache::store('file')->get(
                            $cacheKey
                        );

                        if (
                            is_string($cachedTokenAfterLock)
                            && trim($cachedTokenAfterLock) !== ''
                        ) {
                            return [
                                'token' => trim($cachedTokenAfterLock),
                                'source' => 'cache',
                            ];
                        }

                        $token = $this->authenticationService->authenticate(
                            environment: $environment,
                            privateKeyPem: $privateKeyPem,
                            certificateBase64: $certificateBase64,
                            modulusBase64: $modulusBase64,
                            exponentBase64: $exponentBase64,
                        );

                        Cache::store('file')->put(
                            $cacheKey,
                            $token,
                            now()->addMinutes(
                                $this->cacheTtlMinutes()
                            )
                        );

                        return [
                            'token' => $token,
                            'source' => 'authentication',
                        ];
                    }
                );
        } catch (LockTimeoutException $exception) {
            $cachedTokenAfterTimeout = Cache::store('file')->get(
                $cacheKey
            );

            if (
                is_string($cachedTokenAfterTimeout)
                && trim($cachedTokenAfterTimeout) !== ''
            ) {
                return [
                    'token' => trim($cachedTokenAfterTimeout),
                    'source' => 'cache',
                ];
            }

            throw new RuntimeException(
                'No fue posible obtener el lock para generar el token SII de boleta y el cache continúa vacío.',
                0,
                $exception
            );
        }

    }

        string $environment,
        int $companyId,
        int $certificateId,

        return sprintf(

    }
