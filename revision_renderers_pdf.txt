    public function render(
        DtePrintablePackage $package
    ): DtePdfDocumentRenderResult {
        if ($package->format() !== PrintFormat::LETTER) {
            throw new \DomainException(
                'DompdfDteLetterRenderer solo admite formato Carta.'
            );
        }

        $options = new Options();

        $options->set(
            'defaultFont',
            'DejaVu Sans'
        );

        $options->set(
            'isRemoteEnabled',
            false
        );

        $options->set(
            'isPhpEnabled',
            false
        );

        $options->set(
            'isJavascriptEnabled',
            false
        );

        /*
         * Se crea una instancia nueva para cada render.
         * Dompdf no debe reutilizarse para documentos distintos.
         */
        $dompdf = new Dompdf(
            $options
        );

        $dompdf->setPaper(
            'letter',
            'portrait'
        );

        $dompdf->loadHtml(
            $this->buildHtml(
                $package
            ),
            'UTF-8'
        );

        $dompdf->render();

        $pdfBytes =
            $dompdf->output();

        $pageCount =
            $dompdf
                ->getCanvas()
                ->get_page_count();

        return new DtePdfDocumentRenderResult(
            pdfBytes:
                $pdfBytes,
            format:
                $package->format(),
            pageCount:
                $pageCount,
        );
    }

    private function buildHtml(
        DtePrintablePackage $package
    ): string {
        $physicalCopies = [];

        foreach ($package->copies() as $copy) {
            for (
                $copyNumber = 1;
                $copyNumber <= $copy->quantity();
                $copyNumber++
            ) {
                $physicalCopies[] =
                    $this->buildPhysicalCopy(
                        package:
                            $package,
                        copy:
                            $copy,
                    );
            }
        }

        $pages = [];

        $lastIndex =
            count($physicalCopies) - 1;

        foreach (
            $physicalCopies as
            $index => $physicalCopy
        ) {
            $pages[] =
                '<section class="dte-page">'
                . $physicalCopy
                . '</section>';

            if ($index !== $lastIndex) {
                $pages[] =
                    '<div class="page-break"></div>';
            }
        }

        return
            '<!DOCTYPE html>'
            . '<html lang="es">'
            . '<head>'
            . '<meta charset="UTF-8">'
            . '<style>'
            . $this->css()
            . '</style>'
            . '</head>'
            . '<body>'
            . implode(
                '',
                $pages
            )
            . '</body>'
            . '</html>';
    }

    /**
     * @param DtePrintableLineItem[] $items
     */
    private function buildItemsTable(
        array $items
    ): string {
        $html =
            '<table class="items-table">'
            . '<thead>'
            . '<tr>'
            . '<th class="col-code">CÃ³digo</th>'
            . '<th>DescripciÃ³n</th>'
            . '<th class="col-qty">Cant.</th>'
            . '<th class="col-money">P. Unit.</th>'
            . '<th class="col-discount">Desc.</th>'
            . '<th class="col-money">Total</th>'
            . '</tr>'
            . '</thead>'
            . '<tbody>';

        foreach ($items as $item) {
            $code = '';

            if (
                $item->itemCode() !== null
                && trim($item->itemCode()) !== ''
            ) {
                $code =
                    $item->itemCode();

                if (
                    $item->itemCodeType() !== null
                    && trim(
                        $item->itemCodeType()
                    ) !== ''
                ) {
                    $code =
                        $item->itemCodeType()
                        . ': '
                        . $code;
                }
            }

            $description =
                $item->name();

            if (
                $item->description() !== null
                && trim(
                    $item->description()
                ) !== ''
            ) {
                $description .=
                    '<br><span class="item-description">'
                    . $this->escape(
                        $item->description()
                    )
                    . '</span>';
            }

            if ($item->taxExempt()) {
                $description .=
                    '<br><span class="exempt-label">EXENTO</span>';
            }

            $discount = '-';

            if (
                $item->discountPercent() != 0.0
                || $item->discountAmount() != 0.0
            ) {
                $parts = [];

                if (
                    $item->discountPercent()
                    != 0.0
                ) {
                    $parts[] =
                        $this->decimal(
                            $item->discountPercent()
                        )
                        . '%';
                }

                if (
                    $item->discountAmount()
                    != 0.0
                ) {
                    $parts[] =
                        '$ '
                        . $this->money(
                            $item->discountAmount()
                        );
                }

                $discount =
                    implode(
                        '<br>',
                        $parts
                    );
            }

            $html .=
                '<tr>'
                . '<td>'
                . $this->escape(
                    $code
                )
                . '</td>'
                . '<td>'
                . $this->escape(
                    $item->name()
                );

            if (
                $item->description() !== null
                && trim(
                    $item->description()
                ) !== ''
            ) {
                $html .=
                    '<br><span class="item-description">'
                    . $this->escape(
                        $item->description()
                    )
                    . '</span>';
            }

            if ($item->taxExempt()) {
                $html .=
                    '<br><span class="exempt-label">EXENTO</span>';
            }

            $html .=
                '</td>'
                . '<td class="number">'
                . $this->decimal(
                    $item->quantity()
                )
                . '</td>'
                . '<td class="number">$ '
                . $this->money(
                    $item->unitPrice()
                )
                . '</td>'
                . '<td class="number">'
                . $discount
                . '</td>'
                . '<td class="number">$ '
                . $this->money(
                    $item->lineAmount()
                )
                . '</td>'
                . '</tr>';
        }

        return
            $html
            . '</tbody>'
            . '</table>';
    }

    private function buildPdf417(
        DtePrintablePackage $package
    ): string {
        $selection =
            $package->pdf417();

        $renderResult =
            $selection->renderResult();

        $metrics =
            $selection->physicalMetrics();

        $svgDataUri =
            'data:'
            . $renderResult->mimeType()
            . ';base64,'
            . base64_encode(
                $renderResult->svg()
            );

        $symbolWidth =
            $this->cssNumber(
                $metrics->symbolWidthMm()
            );

        $symbolHeight =
            $this->cssNumber(
                $metrics->symbolHeightMm()
            );

        $quietZone =
            $this->cssNumber(
                $metrics->quietZoneMm()
            );

        return
            '<div class="ted-section">'
            . '<div class="barcode-wrapper" style="'
            . 'padding:'
            . $quietZone
            . 'mm;'
            . '">'
            . '<img'
            . ' alt="PDF417 TED"'
            . ' src="'
            . $svgDataUri
            . '"'
            . ' style="display:block;'
            . 'width:'
            . $symbolWidth
            . 'mm;'
            . 'height:'
            . $symbolHeight
            . 'mm;"'
            . '>'
            . '</div>'
            . '<div class="ted-caption">'
            . 'Timbre ElectrÃ³nico SII'
            . '</div>'
            . '</div>';
    }

        if (!$this->isSupportedLogoDataUri($logoDataUri)) {
            return '';

        return
            '<div style="margin:0 0 3mm 0;">'

    private function isSupportedLogoDataUri(
        ?string $logoDataUri
    ): bool {
        if (
            $logoDataUri === null
            || trim($logoDataUri) === ''
        ) {
            return false;
        }

        return
            str_starts_with(
                $logoDataUri,
                'data:image/png;base64,'
            )
            || str_starts_with(
                $logoDataUri,
                'data:image/jpeg;base64,'
            );
    }

    private function siiOfficeLabel(
        ?string $officeName
    ): string {
        if (
            $officeName === null
            || trim($officeName) === ''
        ) {
            return 'S.I.I.';
        }

        return
            'S.I.I. - '
            . mb_strtoupper(
                trim($officeName),
                'UTF-8'
            );
    }
    private function formatDate(
        ?string $date
    ): string {
        if (
            $date === null
            || trim($date) === ''
        ) {
            return '';
        }

        $parsed =
            \DateTimeImmutable::createFromFormat(
                'Y-m-d',
                $date
            );

        if ($parsed === false) {
            return $date;
        }

        return
            $parsed->format(
                'd-m-Y'
            );
    }

    private function money(
        float $value
    ): string {
        return number_format(
            $value,
            0,
            ',',
            '.'
        );
    }

    private function decimal(
        float $value
    ): string {
        $formatted =
            number_format(
                $value,
                6,
                ',',
                '.'
            );

        return rtrim(
            rtrim(
                $formatted,
                '0'
            ),
            ','
        );
    }

    private function cssNumber(
        float $value
    ): string {
        return rtrim(
            rtrim(
                number_format(
                    $value,
                    5,
                    '.',
                    ''
                ),
                '0'
            ),
            '.'
        );
    }

    private function escape(
        ?string $value
    ): string {
        return htmlspecialchars(
            $value ?? '',
            ENT_QUOTES
            | ENT_SUBSTITUTE,
            'UTF-8'
        );
    }

    private function calculatePageHeightMm(
        DtePrintablePackage $package
    ): float {
        $document =
            $package->document();
        $hasLogo =
            $document
                ->issuer()
                ->logoDataUri() !== null;

        $pdf417HeightMm =
            $package
                ->pdf417()
                ->physicalMetrics()
                ->totalHeightMm();

        $hasCedible = false;

        foreach (
            $package->copies()
            as $copy
        ) {
            if (
                $copy->type()
                === PrintCopyType::CEDIBLE
            ) {
                $hasCedible = true;
                break;
            }
        }

        /*
        * El total en palabras agrega altura en todas las copias.
        * El Acuse de Recibo agrega altura sÃ³lo a CEDIBLE.
        *
        * Dompdf usa un Ãºnico tamaÃ±o de pÃ¡gina para todo el PDF,
        * por lo que calculamos la altura necesaria para la copia
        * fÃ­sicamente mÃ¡s larga.

        $estimatedHeightMm =
